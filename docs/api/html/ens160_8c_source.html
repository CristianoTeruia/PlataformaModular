99static HAL_StatusTypeDef EnterStandardMode(bool with_reset)
100{
101 HAL_StatusTypeDef status;
102
103 if (with_reset)
104 {
105 status = RegWrite(ENS160_REG_OPMODE, ENS160_OPMODE_RESET);
106 if (status != HAL_OK) return status;
107 ens160_reset_count++;
108 App_Delay(ENS160_BOOT_MS);
109 }
110 status = RegWrite(ENS160_REG_OPMODE, ENS160_OPMODE_IDLE);
111 if (status != HAL_OK) return status;
112 App_Delay(ENS160_BOOT_MS);
113
114 status = RegWrite(ENS160_REG_COMMAND, ENS160_CMD_NOP);
115 if (status != HAL_OK) return status;
116 status = RegWrite(ENS160_REG_COMMAND, ENS160_CMD_CLRGPR);
117 if (status != HAL_OK) return status;
118 App_Delay(ENS160_BOOT_MS);
119
120 /* Le a versao do firmware do sensor (como o ESPHome). So diagnostico: falha aqui nao interrompe. */
121 if (RegWrite(ENS160_REG_COMMAND, ENS160_CMD_GET_APPVER) == HAL_OK)
122 {
123 uint8_t version[3];
124 App_Delay(ENS160_BOOT_MS);
125 if (RegRead(ENS160_REG_GPR_READ_4, version, 3U) == HAL_OK)
126 {
127 ens160_fw_major = version[0];
128 ens160_fw_minor = version[1];
129 ens160_fw_build = version[2];
130 }
131 }
132
133 status = RegWrite(ENS160_REG_OPMODE, ENS160_OPMODE_STANDARD);
134 if (status != HAL_OK) return status;
135 App_Delay(ENS160_MODE_MS);
136
137 ens160_init_count++;
138 return HAL_OK;
139}

152HAL_StatusTypeDef ENS160_Read(bool compensation_valid, float temperature_c, float humidity_pct,
153 ENS160_Sample *out)
154{
155 static const uint8_t candidates[2] = { ENS160_ADDR_HIGH, ENS160_ADDR_LOW };
156 uint8_t data[5];
157 uint8_t status_reg = 0U;
158 HAL_StatusTypeDef status = HAL_ERROR;
159 bool fresh_init = false;
160 uint8_t max_steps;
161
162 if (out == NULL) return HAL_ERROR;
163
164 /* Detecta o sensor (confirmando o PART_ID) enquanto ele nao foi encontrado. */
165 if (ens160_addr == 0U)
166 {
167 for (uint8_t i = 0U; i < 2U; i++)
168 {
169 ens160_addr = candidates[i];
170 status = RegRead(ENS160_REG_PART_ID, data, 2U);
171 if (status == HAL_OK)
172 {
173 ens160_part_id = (uint16_t)((uint16_t)data[0] | ((uint16_t)data[1] << 8));
174 if (ens160_part_id == ENS160_PART_ID_EXPECTED) break;
175 }
176 ens160_addr = 0U;
177 }
178 if (ens160_addr == 0U)
179 {
180 ens160_last_hal = (uint32_t)status;
181 return HAL_ERROR;
182 }
183 }
184
185 /* Coloca o sensor em STANDARD somente se ele NAO esta medindo. Um sensor que ja esta em STANDARD
186 * (inclusive apos reiniciar a placa) nao e tocado. A 1a tentativa nao usa RESET; depois de uma
187 * tentativa sem sucesso, a proxima usa RESET. */
188 status = RegRead(ENS160_REG_OPMODE, data, 1U);
189 if (status != HAL_OK) return CommFail(status);
190 ens160_opmode = data[0];
191 if ((data[0] != ENS160_OPMODE_STANDARD) || need_init)
192 {
193 status = EnterStandardMode(init_failures > 0U);
194 if (status != HAL_OK) return CommFail(status);
195 need_init = false;
196 fresh_init = true;
197 }
198
199 /* Compensacao com a temperatura e umidade medidas (AHT): T = (C + 273,15) x 64, RH = % x 512. */
200 if (compensation_valid)
201 {
202 float t = temperature_c;
203 float rh = humidity_pct;
204 uint16_t t_raw, rh_raw;
205 if (t < -40.0f) t = -40.0f;
206 if (t > 85.0f) t = 85.0f;
207 if (rh < 0.0f) rh = 0.0f;
208 if (rh > 100.0f) rh = 100.0f;
209 t_raw = (uint16_t)((t + 273.15f) * 64.0f);
210 rh_raw = (uint16_t)(rh * 512.0f);
211 data[0] = (uint8_t)(t_raw & 0xFFU);
212 data[1] = (uint8_t)(t_raw >> 8);
213 data[2] = (uint8_t)(rh_raw & 0xFFU);
214 data[3] = (uint8_t)(rh_raw >> 8);
215 status = HAL_I2C_Mem_Write(&hi2c1, (uint16_t)(ens160_addr << 1), ENS160_REG_TEMP_IN,
216 I2C_MEMADD_SIZE_8BIT, data, 4U, ENS160_TIMEOUT_MS);
217 if (status != HAL_OK) return CommFail(status);
218 }
219
220 /* Aguarda dado novo (NEWDAT). Em regime normal ele ja esta pronto; logo apos iniciar o modo
221 * STANDARD a primeira medicao leva cerca de 1 s ou mais. A espera cede a CPU.
222 * O bit STATAS (0x80) NAO e usado como criterio: neste modulo ele permanece desligado mesmo com
223 * o sensor medindo (DEVICE_STATUS = 0x03). */
224 max_steps = fresh_init ? ENS160_FIRST_DATA_STEPS : ENS160_DATA_STEPS;
225 for (uint8_t step = 0U; ; step++)
226 {
227 status = RegRead(ENS160_REG_STATUS, &status_reg, 1U);
228 if (status != HAL_OK) return CommFail(status);
229 ens160_last_status = status_reg;
230 ens160_last_hal = (uint32_t)HAL_OK;
231
232 if ((status_reg & ENS160_STATUS_STATER) != 0U)
233 {
234 /* Erro interno do sensor: tenta de novo, desta vez com RESET. */
235 ens160_error_count++;
236 need_init = true;
237 if (init_failures < 255U) init_failures++;
238 return HAL_ERROR;
239 }
240 if ((status_reg & ENS160_STATUS_NEWDAT) != 0U) break;
241 if (step >= max_steps)
242 {
243 ReadEcho();
244 ens160_timeout_count++;
245 if (fresh_init && (init_failures < 255U)) init_failures++; /* proxima inicializacao com RESET */
246 if (++timeout_streak >= ENS160_STUCK_CYCLES)
247 {
248 /* Medindo ha varios ciclos sem entregar dado: reinicializa com RESET. */
249 need_init = true;
250 if (init_failures < 255U) init_failures++;
251 timeout_streak = 0U;
252 }
253 return HAL_BUSY;
254 }
255 App_Delay(ENS160_DATA_STEP_MS);
256 }
257
258 status = RegRead(ENS160_REG_DATA_AQI, data, 5U);
259 if (status != HAL_OK) return CommFail(status);
260
261 out->aqi = (uint8_t)(data[0] & 0x07U);
262 out->tvoc_ppb = (uint16_t)((uint16_t)data[1] | ((uint16_t)data[2] << 8));
263 out->eco2_ppm = (uint16_t)((uint16_t)data[3] | ((uint16_t)data[4] << 8));
264 out->validity = (uint8_t)((status_reg >> 2) & 0x03U);
265 ReadEcho();
266
267 if ((out->aqi == 0U) && (out->tvoc_ppb == 0U) && (out->eco2_ppm == 0U))
268 {
269 /* Nao e medicao: um ENS160 em operacao informa eCO2 >= 400 ppm e AQI de 1 a 5, mesmo em aquecimento.
270 * Trata como "sem dado" (nao e gravado) e conta para uma eventual reinicializacao. */
271 ens160_zero_count++;
272 if (fresh_init && (init_failures < 255U)) init_failures++;
273 if (++timeout_streak >= ENS160_STUCK_CYCLES)
274 {
275 need_init = true;
276 if (init_failures < 255U) init_failures++;
277 timeout_streak = 0U;
278 }
279 return HAL_BUSY;
280 }
281
282 init_failures = 0U;
283 timeout_streak = 0U;
284 ens160_last_aqi = out->aqi;
285 ens160_last_tvoc = out->tvoc_ppb;
286 ens160_last_eco2 = out->eco2_ppm;
287 ens160_last_validity = out->validity;
288 return HAL_OK;
289}
