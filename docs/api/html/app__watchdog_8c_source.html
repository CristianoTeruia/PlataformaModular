55void Watchdog_Service(void)
56{
57 uint32_t now = HAL_GetTick();
58 uint8_t stale = 0U;
59
60 wdg_last_checkin[WDG_TASK_COMMAND] = now;
61 for (uint32_t i = 1U; i < (uint32_t)WDG_TASK_COUNT; i++)
62 {
63 if ((uint32_t)(now - wdg_last_checkin[i]) > wdg_limit_ms[i]) stale |= (uint8_t)(1U << i);
64 }
65 wdg_stale_mask = stale;
66
67 if ((stale == 0U) && ((uint32_t)(now - wdg_last_feed) >= 100U))
68 {
69 (void)HAL_IWDG_Refresh(&hiwdg);
70 wdg_last_feed = now;
71 }
