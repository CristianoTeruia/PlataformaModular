54/* Protege o barramento fisico I2C2 compartilhado pelo OLED e pelas EEPROMs.
55 * Antes do escalonador, as inicializacoes ocorrem sequencialmente e nao precisam
56 * do mutex. Durante o RTOS, todas as transacoes protegidas usam este lock. */
57HAL_StatusTypeDef I2C2_BusAcquire(void)
58{
59 osKernelState_t state = osKernelGetState();
60 if ((state == osKernelRunning) || (state == osKernelReady))
61 {
62 if (i2c2MutexHandle == NULL) return HAL_ERROR;
63 return (osMutexAcquire(i2c2MutexHandle, osWaitForever) == osOK) ? HAL_OK : HAL_ERROR;
64 }
65 return HAL_OK;
66}

68/* Protege HAL_RTC_Get/Set entre tasks. Mutex "folha": nunca e segurado enquanto se
69 * espera por outro mutex, portanto nao cria deadlock. Antes do RTOS nao faz nada. */

71{
72 if ((osKernelGetState() == osKernelRunning) && (rtcMutexHandle != NULL))

127/* Recupera o I2C2 (OLED + EEPROMs): libera uma linha possivelmente presa por um escravo
128 * (ate 9 pulsos de SCL + STOP), reinicia o periferico (HAL_I2C_Init faz SWRST) e
129 * reinicializa o SSD1306. Segura o mutex do I2C2 durante todo o processo. */
131static void I2C2_BitDelay(void)
132{
133 for (volatile uint32_t i = 0U; i < 60U; i++) { __NOP(); }
134}

145static HAL_StatusTypeDef RecoverI2C2AndOled(void)
146{
147 GPIO_InitTypeDef g = {0};
148 HAL_StatusTypeDef st;
149
150 if (I2C2_BusAcquire() != HAL_OK) return HAL_ERROR;
151
152 (void)HAL_I2C_DeInit(&hi2c2);
153
154 g.Mode = GPIO_MODE_OUTPUT_OD;
155 g.Pull = GPIO_NOPULL;
156 g.Speed = GPIO_SPEED_FREQ_LOW;
157 g.Pin = MEMORY_SCL_Pin; HAL_GPIO_Init(MEMORY_SCL_GPIO_Port, &g);
158 g.Pin = MEMORY_SDA_Pin; HAL_GPIO_Init(MEMORY_SDA_GPIO_Port, &g);
159 HAL_GPIO_WritePin(MEMORY_SDA_GPIO_Port, MEMORY_SDA_Pin, GPIO_PIN_SET);
160 HAL_GPIO_WritePin(MEMORY_SCL_GPIO_Port, MEMORY_SCL_Pin, GPIO_PIN_SET);
161 I2C2_BitDelay();
162
163 for (uint8_t i = 0U; i < 9U; i++)
164 {
165 if (HAL_GPIO_ReadPin(MEMORY_SDA_GPIO_Port, MEMORY_SDA_Pin) == GPIO_PIN_SET) break;
166 HAL_GPIO_WritePin(MEMORY_SCL_GPIO_Port, MEMORY_SCL_Pin, GPIO_PIN_RESET); I2C2_BitDelay();
167 HAL_GPIO_WritePin(MEMORY_SCL_GPIO_Port, MEMORY_SCL_Pin, GPIO_PIN_SET); I2C2_BitDelay();
168 }
169
170 /* Condicao de STOP: SDA sobe com SCL alto. */
171 HAL_GPIO_WritePin(MEMORY_SDA_GPIO_Port, MEMORY_SDA_Pin, GPIO_PIN_RESET); I2C2_BitDelay();
172 HAL_GPIO_WritePin(MEMORY_SCL_GPIO_Port, MEMORY_SCL_Pin, GPIO_PIN_SET); I2C2_BitDelay();
173 HAL_GPIO_WritePin(MEMORY_SDA_GPIO_Port, MEMORY_SDA_Pin, GPIO_PIN_SET); I2C2_BitDelay();
174
175 HAL_GPIO_DeInit(MEMORY_SCL_GPIO_Port, MEMORY_SCL_Pin);
176 HAL_GPIO_DeInit(MEMORY_SDA_GPIO_Port, MEMORY_SDA_Pin);
177
178 st = HAL_I2C_Init(&hi2c2); /* MSP reconfigura os pinos como I2C2 */
179 if (st == HAL_OK)
180 {
181 st = SSD1306_Init();
182 }
183
184 i2c2_recovery_count++;
185 i2c2_recovery_status = (uint32_t)st;
186 I2C2_BusRelease();
187 return st;
188}

426/* Tarefa de aquisição: mantém o período de 10 s usando o tick do RTOS.
427 * O acesso às variáveis de sensores é protegido contra leitura simultânea pelo OLED.
428 * O mutex de EEPROM serializa gravação, exportação e limpeza. */
429/* Atualização de baixa prioridade. Copia o estado compartilhado sob mutexes curtos;
430 * a renderização e as transações I2C2 ocorrem depois de liberar os mutexes. */
431void StartOledTask(void *argument)
432{
433 (void)argument;
434 uint8_t button_count = 0U, usb_prev = 0U;
435 OledPower_Init(&oled_power_state, (uint32_t)((uint64_t)osKernelGetTickCount() * 1000U / osKernelGetTickFreq()));
436 uint32_t next_page_tick = osKernelGetTickCount() + (3000U * osKernelGetTickFreq()) / 1000U;
437 uint32_t last_recovery_tick = 0U;
438 for (;;)
439 {
440 uint32_t now = osKernelGetTickCount();
441 oled_task_loops++;
442 Watchdog_Checkin(WDG_TASK_OLED);
443
444 /* V13: o OLED apaga sozinho (economia de energia) e acende com o botao (PA0, ativo em nivel
445 * baixo, filtrado por 2 leituras seguidas) ou quando uma operacao USB comeca. */
446 {
447 uint32_t now_ms = (uint32_t)((uint64_t)now * 1000U / osKernelGetTickFreq());
448 uint8_t raw = (HAL_GPIO_ReadPin(BUTTON_GPIO_Port, BUTTON_Pin) == GPIO_PIN_RESET) ? 1U : 0U;
449 uint8_t usb_active = (usb_operation_state != 0U) ? 1U : 0U;
450 OledPowerAction action;
451
452 button_count = (raw != 0U) ? ((button_count < 255U) ? (uint8_t)(button_count + 1U) : 255U) : 0U;
453 action = OledPower_Update(&oled_power_state, now_ms, (button_count >= 2U) ? 1U : 0U,
454 ((usb_active != 0U) && (usb_prev == 0U)) ? 1U : 0U, OLED_AUTO_OFF_MS);
455 usb_prev = usb_active;
456
457 if (action == OLED_POWER_TURN_ON)
458 {
459 if (SSD1306_SetPower(true) == HAL_OK) { oled_on_count++; oledRefreshRequested = 1U; }
460 else OledPower_ActionFailed(&oled_power_state, action, now_ms);
461 }
462 else if (action == OLED_POWER_TURN_OFF)
463 {
464 if (SSD1306_SetPower(false) == HAL_OK) oled_off_count++;
465 else OledPower_ActionFailed(&oled_power_state, action, now_ms);
466 }
467 }
468 if (oled_power_state.on == 0U)
469 {
470 /* Display apagado: nao renderiza nem ocupa o I2C2; so verifica o botao a cada 20 ms. */
471 osDelay(20U);
472 continue;
473 }
474
475 if ((int32_t)(now - next_page_tick) >= 0)
476 {
477 display_page = (display_page == 0U) ? 1U : 0U;
478 oled_page_switch_count++;
479 next_page_tick = now + (3000U * osKernelGetTickFreq()) / 1000U;
480 oledRefreshRequested = 1U;
481 }
482
483 if (oledRefreshRequested != 0U)
484 {

486 oledRefreshRequested = 0U;
487
488 oled_task_stage = 1U;
489 if (osMutexAcquire(sensorDataMutexHandle, osWaitForever) == osOK)
490 {

492 snapshot.sensor_detected_mask = Sensors_DetectedMask();
493 snapshot.rtc_time_valid = rtc_time_valid;
494 snapshot.ssd1306_status = ssd1306_status;
495 (void)osMutexRelease(sensorDataMutexHandle);
496 }

499 * Diagnostico V5: a exportacao pode manter eepromMutexHandle ocupado
500 * por bastante tempo. A tarefa OLED nao deve ficar bloqueada
501 * indefinidamente: se nao conseguir obter o mutex em ate ~100 ms,
502 * preserva a tela atual e tenta atualizar novamente no proximo ciclo.
503 */
504 uint32_t eeprom_wait_ticks =
505 (100U * osKernelGetTickFreq() + 999U) / 1000U;
506 if (eeprom_wait_ticks == 0U) eeprom_wait_ticks = 1U;
507
508 oled_task_stage = 2U;

510 {
511 oledRefreshRequested = 1U;
512 osDelay(20U);
513 continue;
514 }

517 snapshot.eeprom_metadata_load_ok = eeprom_metadata_load_ok;
518 snapshot.eeprom_full = eeprom_full;
519 snapshot.eeprom_active = eeprom_metadata.reserved;
520 snapshot.eeprom_next_address = eeprom_next_address;
521 snapshot.usb_operation_state = usb_operation_state;
522 snapshot.diagnostic_sensor_stage = diagnostic_sensor_stage;
523 snapshot.diagnostic_export_stage = diagnostic_export_stage;
524 snapshot.diagnostic_export_result = diagnostic_export_result;
525 snapshot.diagnostic_usb_tx_result = diagnostic_usb_tx_result;

527
528 snapshot.oled_available = oled_available;
529 snapshot.display_page = display_page;
530 oled_task_stage = 3U;
531 UpdateDisplaySnapshot(&snapshot);
532 oled_task_stage = 0U;
533
534 /* V9: uma atualizacao falhando => recupera o I2C2 e o SSD1306
535 * (no maximo uma vez a cada 5 s). */
536 if ((oled_consecutive_fail >= 1U) &&
537 ((last_recovery_tick == 0U) ||
538 ((uint32_t)(osKernelGetTickCount() - last_recovery_tick) >= 5000U)))
539 {

542 last_recovery_tick = osKernelGetTickCount();

552void StartSensorTask(void *argument)
553{
554 (void)argument;
555 uint32_t period_ticks = (SENSOR_ACQUISITION_INTERVAL_MS * osKernelGetTickFreq()) / 1000U;
556 if (period_ticks == 0U)
557 period_ticks = 1U;
558 uint32_t next_wake_tick = osKernelGetTickCount(); /* primeira amostra imediata */
559 uint8_t consecutive_no_sensor_samples = 0U;
560 uint32_t last_sensor_i2c_recovery_tick = 0U;
561
562 for (;;)
563 {
564 uint32_t now;
565
566 Watchdog_Checkin(WDG_TASK_SENSOR);
567#if ENABLE_WDT_TEST
568 if (wdg_test_hang != 0U)
569 {
570 /* Comando WDTTEST: simula task travada; o watchdog deve reiniciar a placa. */
571 for (;;) { (void)osDelay(1000U); }
572 }
573#endif
574
575 (void)osMutexAcquire(sensorDataMutexHandle, osWaitForever);
576 now = HAL_GetTick();
577
578 sensor_cycle_active = 1U;
579 App_RTC_Lock();
580 (void)HAL_RTC_GetTime(&hrtc, &rtc_test_time, RTC_FORMAT_BIN);
581 (void)HAL_RTC_GetDate(&hrtc, &rtc_test_date, RTC_FORMAT_BIN);
582 App_RTC_Unlock();
583
584 /* Le todos os sensores da tabela (sensor_registry.c). Estagios 1..3 como na V10. */
585 for (uint8_t si = 0U; si < Sensors_Count(); si++)
586 {
587 diagnostic_sensor_stage = (uint8_t)(si + 1U);
588 Sensors_ReadOne(si);
589 }
590 Sensors_UpdateDisplayValues();
591
592 diagnostic_sensor_stage = 4U; /* avaliacao e possivel gravacao EEPROM */
593 if (Sensors_AvailableMask() != 0U)
594 {
595 consecutive_no_sensor_samples = 0U;
596 /*
597 * Diagnostico V4: a tarefa de sensores NAO deve esperar indefinidamente
598 * pela EEPROM enquanto mantem sensorDataMutexHandle. Isso poderia
599 * bloquear a atualizacao do OLED e impedir a alternancia do LED.
600 * Timeout em ticks: aproximadamente 250 ms.
601 */
602 uint32_t eeprom_wait_ticks =
603 (250U * osKernelGetTickFreq() + 999U) / 1000U;
604 if (eeprom_wait_ticks == 0U) eeprom_wait_ticks = 1U;
605
606 if (osMutexAcquire(eepromMutexHandle, eeprom_wait_ticks) == osOK)
607 {
608 if (EEPROM_SaveSensorRecord() != HAL_OK && eeprom_status_display == 0U)
609 eeprom_status_display = 1U;
610 (void)osMutexRelease(eepromMutexHandle);
611 }
612 else if ((eeprom_status_display == 0U) && (usb_operation_state == 0U))
613 {
614 /* A amostra atual pode ser perdida; nao altera registros existentes.
615 * Durante EXPORT/CLEAR o mutex fica ocupado de proposito: nao e erro da EEPROM. */
616 eeprom_status_display = 1U;
617 }
618 }
619 else
620 {
621 if (consecutive_no_sensor_samples < 255U)
622 consecutive_no_sensor_samples++;
623
624 /* Recuperação limitada a uma tentativa por 30 s e somente no I2C1. */
625 if ((consecutive_no_sensor_samples >= 3U) &&
626 ((last_sensor_i2c_recovery_tick == 0U) ||
627 ((uint32_t)(now - last_sensor_i2c_recovery_tick) >= 30000U)))
628 {
629 (void)RecoverSensorI2C1();
630 last_sensor_i2c_recovery_tick = now;
631 consecutive_no_sensor_samples = 0U;
632 }
633 }
634
635 HAL_GPIO_TogglePin(LED_GPIO_Port, LED_Pin);
636 diagnostic_sensor_stage = 5U; /* ciclo chegou ao LED */
637 sensor_cycle_active = 0U;
638 (void)osMutexRelease(sensorDataMutexHandle);
639
640 /* Mantém o ciclo periódico; se o processamento ultrapassar o prazo,
641 * agenda o próximo ciclo a partir do instante atual, sem amostras em rajada. */
642 {
643 next_wake_tick += period_ticks;
644 if ((int32_t)(osKernelGetTickCount() - next_wake_tick) >= 0) // ciclo estourou o período
645 next_wake_tick = osKernelGetTickCount() + period_ticks;
646 }
647 (void)osDelayUntil(next_wake_tick);
648 }
649}

655void App_RTOS_Create(void)
