81HAL_StatusTypeDef SSD1306_Init(void)
82{
83 uint8_t command;
84
85 /* Display OFF */
86 command = 0xAE;
87 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
88 I2C_MEMADD_SIZE_8BIT,
89 &command, 1, 100) != HAL_OK)
90 return HAL_ERROR;
91
92 /* Display clock divide ratio / oscillator frequency */
93 command = 0x80;
94 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
95 I2C_MEMADD_SIZE_8BIT,
96 &command, 1, 100) != HAL_OK)
97 return HAL_ERROR;
98
99 /* Multiplex ratio: 64 linhas */
100 command = 0x3F;
101 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
102 I2C_MEMADD_SIZE_8BIT,
103 &command, 1, 100) != HAL_OK)
104 return HAL_ERROR;
105
106 /* Display offset = 0 */
107 command = 0x00;
108 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
109 I2C_MEMADD_SIZE_8BIT,
110 &command, 1, 100) != HAL_OK)
111 return HAL_ERROR;
112
113 /* Start line = 0 */
114 command = 0x40;
115 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
116 I2C_MEMADD_SIZE_8BIT,
117 &command, 1, 100) != HAL_OK)
118 return HAL_ERROR;
119
120 /* Charge Pump: ENABLE */
121 command = 0x8D;
122 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
123 I2C_MEMADD_SIZE_8BIT,
124 &command, 1, 100) != HAL_OK)
125 return HAL_ERROR;
126
127 command = 0x14;
128 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
129 I2C_MEMADD_SIZE_8BIT,
130 &command, 1, 100) != HAL_OK)
131 return HAL_ERROR;
132
133 /* Segment remap */
134 command = 0xA1;
135 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
136 I2C_MEMADD_SIZE_8BIT,
137 &command, 1, 100) != HAL_OK)
138 return HAL_ERROR;
139
140 /* COM scan direction */
141 command = 0xC8;
142 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
143 I2C_MEMADD_SIZE_8BIT,
144 &command, 1, 100) != HAL_OK)
145 return HAL_ERROR;
146
147 /* COM pins hardware configuration */
148 command = 0x12;
149 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
150 I2C_MEMADD_SIZE_8BIT,
151 &command, 1, 100) != HAL_OK)
152 return HAL_ERROR;
153
154 /* Contrast */
155 command = 0x7F;
156 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
157 I2C_MEMADD_SIZE_8BIT,
158 &command, 1, 100) != HAL_OK)
159 return HAL_ERROR;
160
161 /* Pre-charge period */
162 command = 0xD9;
163 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
164 I2C_MEMADD_SIZE_8BIT,
165 &command, 1, 100) != HAL_OK)
166 return HAL_ERROR;
167
168 command = 0xF1;
169 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
170 I2C_MEMADD_SIZE_8BIT,
171 &command, 1, 100) != HAL_OK)
172 return HAL_ERROR;
173
174 /* VCOMH deselect level */
175 command = 0xDB;
176 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
177 I2C_MEMADD_SIZE_8BIT,
178 &command, 1, 100) != HAL_OK)
179 return HAL_ERROR;
180
181 command = 0x40;
182 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
183 I2C_MEMADD_SIZE_8BIT,
184 &command, 1, 100) != HAL_OK)
185 return HAL_ERROR;
186
187 /* Entire display follows RAM */
188 command = 0xA4;
189 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
190 I2C_MEMADD_SIZE_8BIT,
191 &command, 1, 100) != HAL_OK)
192 return HAL_ERROR;
193
194 /* Normal display */
195 command = 0xA6;
196 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
197 I2C_MEMADD_SIZE_8BIT,
198 &command, 1, 100) != HAL_OK)
199 return HAL_ERROR;
200
201 /* Display ON */
202 command = 0xAF;
203 if (HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
204 I2C_MEMADD_SIZE_8BIT,
205 &command, 1, 100) != HAL_OK)
206 return HAL_ERROR;
207
208 App_Delay(100);
209
210 return HAL_OK;
211}

214/* Sequencias do datasheet do SSD1306: desligar = display OFF e bomba de carga desabilitada;
215 * ligar = bomba de carga habilitada e display ON. O conteudo da RAM do display e mantido. */
216HAL_StatusTypeDef SSD1306_SetPower(bool on)
217{
218 static const uint8_t sequence_on[3] = { 0x8D, 0x14, 0xAF };
219 static const uint8_t sequence_off[3] = { 0xAE, 0x8D, 0x10 };
220 const uint8_t *sequence = on ? sequence_on : sequence_off;
221 HAL_StatusTypeDef status = I2C2_BusAcquire();
222
223 if (status != HAL_OK) return status;
224 for (uint8_t i = 0U; i < 3U; i++)
225 {
226 uint8_t command = sequence[i];
227 status = HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00, I2C_MEMADD_SIZE_8BIT, &command, 1, 100);
228 if (status != HAL_OK) break;
229 }
230 I2C2_BusRelease();
231 return status;
232}

262HAL_StatusTypeDef SSD1306_Fill(void)
263{
264 uint8_t buffer[129];
265 HAL_StatusTypeDef status = I2C2_BusAcquire();
266 if (status != HAL_OK) return status;
267
268 /* Primeiro byte indica que os próximos dados são pixels. */
269 buffer[0] = 0x40;
270 for (int i = 1; i < 129; i++) buffer[i] = 0xFF;
271
272 /* Mantem o mutex em toda a operacao, incluindo comandos e pixels. */
273 for (uint8_t page = 0; page < 8; page++)
274 {
275 uint8_t commands[3] = { (uint8_t)(0xB0 | page), 0x00, 0x10 };
276 status = HAL_I2C_Mem_Write(&hi2c2, 0x3D << 1, 0x00,
277 I2C_MEMADD_SIZE_8BIT, commands, 3, 100);
278 if (status != HAL_OK) break;
279
280 status = HAL_I2C_Master_Transmit(&hi2c2, 0x3D << 1, buffer, 129, 100);
281 if (status != HAL_OK) break;
282 }
283
284 I2C2_BusRelease();
285 return status;
286}

313void SSD1306_DrawChar(uint8_t x, uint8_t y, char c)
314{
315 uint8_t index;
316
317 if (c >= '0' && c <= '9')
318 {
319 index = c - '0';
320 }
321 else if (c >= 'A' && c <= 'Z')
322 {
323 index = 10 + (c - 'A');
324 }
325 else if (c == '.')
326 {
327 index = 36;
328 }
329 else if (c == ':')
330 {
331 index = 37;
332 }
333 else if (c == '%')
334 {
335 index = 38;
336 }
337 else if (c == '-')
338 {
339 index = 39;
340 }
341 else
342 {
343 return;
344 }
345
346 for (uint8_t col = 0; col < 5; col++)
347 {
348 uint8_t line = font5x7[index][col];
349
350 for (uint8_t row = 0; row < 7; row++)
351 {
352 if (line & (1 << row))
353 {
354 SSD1306_DrawPixel(x + col, y + row, 1);
355 }
356 }
357 }

391void SSD1306_UpdateScreen(void)
392{
393 uint8_t data[129];
394 HAL_StatusTypeDef status;
395 uint8_t failed_page = 0xFFU;
396 uint8_t failed_stage = 0U; /* 1=comando de pagina; 2=transferencia dos pixels */
397
398 data[0] = 0x40;
399 oled_task_stage = 4U; /* esperando mutex do I2C2 */
400 status = I2C2_BusAcquire();
401 if (status != HAL_OK)
402 {
403 oled_lock_fail_count++;
404 oled_update_fail_count++;
405 oled_consecutive_fail++;
406 oled_failed_page = 0xFFU;
407 oled_failed_stage = 3U; /* falha ao adquirir mutex */
408 ssd1306_status = status;
409 return;
410 }
411 oled_task_stage = 5U; /* transferindo pelo I2C2 */
412
413 for (uint8_t page = 0; page < 8; page++)
414 {
415 uint8_t commands[3];
416 commands[0] = 0xB0 | page;
417 commands[1] = 0x00;
418 commands[2] = 0x10;
419
420 OLED_CaptureI2C2PreState();
421 status = HAL_I2C_Mem_Write(
422 &hi2c2, 0x3D << 1, 0x00, I2C_MEMADD_SIZE_8BIT, commands, 3, 100);
423 if (status != HAL_OK)
424 {
425 failed_page = page;
426 failed_stage = 1U;
427 /* Captura antes de liberar o mutex e antes de outra tarefa poder
428 * iniciar uma transacao que altere o estado da HAL. */
429 oled_i2c_error_snapshot = hi2c2.ErrorCode;
430 oled_i2c_state_snapshot = (uint32_t)hi2c2.State;
431 break;
432 }
433
434 for (uint8_t x = 0; x < 128; x++)
435 {
436 data[x + 1] = ssd1306_buffer[x + page * 128];
437 }
438
439 oled_buffer_checksum_before = OLED_BufferChecksum();
440 OLED_CaptureI2C2PreState();
441 status = HAL_I2C_Master_Transmit(&hi2c2, 0x3D << 1, data, 129, 100);
442 oled_buffer_checksum_after = OLED_BufferChecksum();
443 if (oled_buffer_checksum_before != oled_buffer_checksum_after)
444 oled_buffer_mutation_count++;
445 if (status != HAL_OK)
446 {
447 failed_page = page;
448 failed_stage = 2U;
449 oled_i2c_error_snapshot = hi2c2.ErrorCode;
450 oled_i2c_state_snapshot = (uint32_t)hi2c2.State;
451 break;
452 }
453 }
454
455 I2C2_BusRelease();
456
457 if (status == HAL_OK)
458 {
459 /* Limpa o evento anterior somente apos uma atualizacao completa. */
460 oled_i2c_error_snapshot = 0U;
461 oled_i2c_state_snapshot = (uint32_t)hi2c2.State;
462 oled_failed_page = 0xFFU;
463 oled_failed_stage = 0U;
464 }
465 else
466 {
467 oled_failed_page = failed_page;
468 oled_failed_stage = failed_stage;
469 }
470
471 /* Mantem o resultado da ultima atualizacao para diagnostico. */
472 ssd1306_status = status;
473 if (status == HAL_OK) { oled_update_ok_count++; oled_consecutive_fail = 0U; }
474 else
475 {
476 if (oled_consecutive_fail == 0U)
477 {
478 oled_ff_status = (uint32_t)status;
479 oled_ff_i2cerr = hi2c2.ErrorCode;
480 oled_ff_i2cstate = (uint32_t)hi2c2.State;
481 oled_ff_sensor_active = sensor_cycle_active;
482 }
483 oled_update_fail_count++; oled_consecutive_fail++;
484 }
485}

521void UpdateDisplaySnapshot(const DisplaySnapshot *snapshot)
522{
523 if ((snapshot == NULL) || (snapshot->oled_available == 0U))
524 return;
525
526 if (oled_render_active != 0U)
527 oled_render_overlap_count++;
528 oled_render_active = 1U;
529
530 /* Sensores modulares: o esperado e o que ja foi detectado desde o boot; modulo nao
531 * conectado nao e erro. Erro = sensor detectado que parou de responder. */
532 uint8_t expected = Sensors_PopCount(snapshot->sensor_detected_mask);
533 uint8_t healthy = Sensors_PopCount((uint8_t)(snapshot->sensor_present_mask & snapshot->sensor_detected_mask));
534 const char *missing_sensor = Sensors_FirstMissingName(snapshot->sensor_detected_mask, snapshot->sensor_present_mask);
535 char sensor_error_text[20];
536 const char *overall = "OK";
537 const char *rtc_text = (snapshot->rtc_time_valid != 0U) ? "RTC:VALIDO" : "RTC:INVALIDO";
538 const char *usb_text = "USB:LIVRE";
539 const char *error_text = "ERRO:NENHUM";
540
541 if (snapshot->eeprom_status_display == 1U || snapshot->eeprom_metadata_load_ok == 0U)
542 error_text = "ERRO:EEPROM";
543 else if (expected == 0U)
544 error_text = "ERRO:SEM SENS";
545 else if (missing_sensor != NULL)
546 {
547 (void)snprintf(sensor_error_text, sizeof(sensor_error_text), "ERRO:%s", missing_sensor);
548 error_text = sensor_error_text;
549 }
550 else if (snapshot->ssd1306_status != HAL_OK)
551 error_text = "ERRO:OLED";
552
553 if (healthy < expected || expected == 0U || snapshot->rtc_time_valid == 0U || snapshot->eeprom_status_display != 0U)
554 overall = (snapshot->eeprom_status_display == 1U || snapshot->eeprom_metadata_load_ok == 0U) ? "ERRO" : "AVISO";
555
556 switch (snapshot->usb_operation_state)
557 {
558 case 1U: usb_text = "USB:EXPORTANDO"; break;
559 case 2U: usb_text = "USB:LIMPANDO"; break;
560 case 3U: usb_text = "USB:ERRO"; break;
561 default: usb_text = "USB:LIVRE"; break;
562 }
563
564 SSD1306_Clear();
565 if (snapshot->display_page == 0U)
566 {
567 SSD1306_DrawString(0, 0, "ESTADO DO SISTEMA");
568 SSD1306_DrawString(0, 12, "GERAL:");
569 SSD1306_DrawString(42, 12, overall);
570 SSD1306_DrawString(0, 24, "SENSORES:");
571 SSD1306_DrawUInt(60, 24, healthy);
572 SSD1306_DrawString(69, 24, "/");
573 SSD1306_DrawUInt(75, 24, expected);
574 SSD1306_DrawString(0, 36, rtc_text);
575 SSD1306_DrawString(0, 50, error_text);
576 }
577 else
578 {
579 SSD1306_DrawString(0, 0, "ARMAZENAMENTO");
580 SSD1306_DrawString(0, 12, "EEPROM ATIVA:");
581 if (snapshot->eeprom_metadata_load_ok == 0U)
582 SSD1306_DrawString(78, 12, "ERRO");
583 else if (snapshot->eeprom_full != 0U)
584 SSD1306_DrawString(78, 12, "CHEIA");
585 else if (snapshot->eeprom_active == EEPROM_ACTIVE_EEPROM2)
586 SSD1306_DrawString(78, 12, "EEPROM2");
587 else
588 SSD1306_DrawString(78, 12, "EEPROM1");
589
590 SSD1306_DrawString(0, 24, "PROX. END:");
591 if (snapshot->eeprom_metadata_load_ok != 0U)
592 SSD1306_DrawUInt(60, 24, snapshot->eeprom_next_address);
593 else
594 SSD1306_DrawString(60, 24, "N/D");
595
596 SSD1306_DrawString(0, 38, usb_text);
597 /* V6: linha de diagnostico independente da USB. Valores sao estagios. */
598 SSD1306_DrawString(0, 52, "S:"); SSD1306_DrawUInt(12, 52, snapshot->diagnostic_sensor_stage);
599 SSD1306_DrawString(30, 52, "E:"); SSD1306_DrawUInt(42, 52, snapshot->diagnostic_export_stage);
600 SSD1306_DrawString(60, 52, "T:"); SSD1306_DrawUInt(72, 52, (snapshot->diagnostic_usb_tx_result < 0) ? 9U : (uint8_t)snapshot->diagnostic_usb_tx_result);
601 SSD1306_DrawString(90, 52, "R:"); SSD1306_DrawUInt(102, 52, (snapshot->diagnostic_export_result < 0) ? 9U : (uint8_t)snapshot->diagnostic_export_result);
602 }
603 SSD1306_UpdateScreen();
604 oled_render_active = 0U;
605}
