34static int AppendScaled(char *buffer, size_t capacity, int current,
35 int32_t value, uint32_t scale, uint8_t decimals)
36{
37 uint32_t magnitude;
38 const char *sign = "";
39 int64_t wide_value = value;
40 if (current < 0) return -1;
41 if (wide_value < 0)
42 {
43 sign = "-";
44 wide_value = -wide_value;
45 }
46 magnitude = (uint32_t)wide_value;
47 return snprintf(buffer + current, capacity - (size_t)current,
48 "%s%lu.%0*lu", sign,
49 (unsigned long)(magnitude / scale), (int)decimals,
50 (unsigned long)(magnitude % scale));
51}

54static int Put(char *out, size_t capacity, int pos, int written)
55{
56 if ((pos < 0) || (written < 0)) return -1;
57 if ((size_t)(pos + written) >= capacity) return -1;
58 return pos + written;
59}

138static HAL_StatusTypeDef Mpu_Read(void)
139{
140 HAL_StatusTypeDef st;
141 if (!mpu6500_available)
142 {
143 /* Detecta/inicializa/calibra (o sensor fica acordado) e entao volta a dormir. */
144 if ((MPU6500_Init() == HAL_OK) && (MPU6500_Calibrate() == HAL_OK))
145 {
146 st = MPU6500_Read();
147 (void)MPU6500_Sleep();
148 mpu6500_last_read_ok = (st == HAL_OK) ? 1U : 0U;
149 mpu6500_available = (st == HAL_OK);
150 return st;
151 }
152 mpu6500_last_read_ok = 0U;
153 mpu6500_available = false;
154 return HAL_ERROR;
155 }
156 /* Economia de energia: acorda, le e volta a dormir (o sensor consome alguns mA acordado). */
157 st = MPU6500_Wake();
158 if (st == HAL_OK) st = MPU6500_Read();
159 (void)MPU6500_Sleep();
160 mpu6500_last_read_ok = (st == HAL_OK) ? 1U : 0U;
161 mpu6500_available = (st == HAL_OK);
162 return st;
163}

184static int Mpu_Csv(const uint8_t *p, char *out, size_t cap)
185{
186 static const uint32_t scale[6] = { 1000U, 1000U, 1000U, 100U, 100U, 100U };
187 static const uint8_t decimals[6] = { 3U, 3U, 3U, 2U, 2U, 2U };
188 int pos = 0;
189
190 if (p == NULL) return Put(out, cap, 0, snprintf(out, cap, ",,,,,"));
191 for (uint8_t i = 0U; i < 6U; i++)
192 {
193 if (i > 0U) pos = Put(out, cap, pos, snprintf(out + pos, (pos < 0) ? 0U : cap - (size_t)pos, ","));

195 (int32_t)(int16_t)ReadLE16(&p[2U * i]), scale[i], decimals[i]));
196 }
197 return pos;
198}

287void Sensors_ReadOne(uint8_t index)
288{
289 if (index >= SENSOR_COUNT) return;
290 HAL_StatusTypeDef result = sensor_table[index].read();
291 if (result == HAL_OK)
292 {
293 sensors_available_mask |= sensor_table[index].mask_bit;
294 sensors_detected_mask |= sensor_table[index].mask_bit;
295 }
296 else
297 {
298 sensors_available_mask &= (uint8_t)~sensor_table[index].mask_bit;
299 /* HAL_BUSY = o sensor responde no barramento mas ainda nao entrega dado: conta como
300 * detectado, para o OLED mostrar ERRO:<nome> em vez de ignorar um modulo conectado. */
301 if (result == HAL_BUSY) sensors_detected_mask |= sensor_table[index].mask_bit;
302 }
303}

319uint8_t Sensors_RecordSizeFromMask(uint8_t mask, uint8_t *size)
320{
321 uint8_t known = 0U;
322 uint16_t calculated = 1U + 7U + 2U; /* mascara + RTC + CRC */
323
324 for (uint8_t i = 0U; i < SENSOR_COUNT; i++) known |= sensor_table[i].mask_bit;
325 if ((size == NULL) || (mask == 0U) || ((mask & (uint8_t)~known) != 0U)) return 0U;
326
327 for (uint8_t i = 0U; i < SENSOR_COUNT; i++)
328 {
329 if ((mask & sensor_table[i].mask_bit) != 0U) calculated += sensor_table[i].payload_size;
330 }
331 if (calculated > EEPROM_MAX_RECORD_SIZE) return 0U;
332 *size = (uint8_t)calculated;
333 return 1U;
334}

336/* Grava os dados dos sensores da mascara a partir de buffer[index]; devolve o novo indice.
337 * A mascara deve ter sido validada por Sensors_RecordSizeFromMask(). */
338uint8_t Sensors_Serialize(uint8_t mask, uint8_t *buffer, uint8_t index)
339{
340 for (uint8_t i = 0U; i < SENSOR_COUNT; i++)
341 {
342 if ((mask & sensor_table[i].mask_bit) != 0U)
343 {
344 sensor_table[i].serialize(&buffer[index]);
345 index = (uint8_t)(index + sensor_table[i].payload_size);
346 }
347 }
348 return index;
349}

366int Sensors_CsvFormatRow(uint8_t mask, const uint8_t *payload, char *out, size_t capacity)
367{
368 int pos = 0;
369 for (uint8_t i = 0U; i < SENSOR_COUNT; i++)
370 {
371 const uint8_t *p = ((mask & sensor_table[i].mask_bit) != 0U) ? payload : NULL;
372 int n;
373 if (i > 0U)
374 {
375 if ((size_t)(pos + 1) >= capacity) return -1;
376 out[pos++] = ',';
377 }
378 n = sensor_table[i].csv_format(p, out + pos, capacity - (size_t)pos);
379 if (n < 0) return -1;
380 pos += n;
381 if (p != NULL) payload += sensor_table[i].payload_size;
382 }
383 if ((size_t)pos >= capacity) return -1;
384 out[pos] = '\0';
385 return pos;
386}
